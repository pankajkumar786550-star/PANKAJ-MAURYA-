<script>
  let records = JSON.parse(localStorage.getItem('records')) || [];

  function addRecord() {
    const month = document.getElementById('month').value;
    const name = document.getElementById('name').value;
    const room = document.getElementById('room').value;
    const prev = Number(document.getElementById('prevReading').value);
    const curr = Number(document.getElementById('currReading').value);
    const rate = Number(document.getElementById('rate').value);

    if (!month || !name || !room || curr < prev || isNaN(rate)) {
      alert('Enter valid monthly details');
      return;
    }

    const units = curr - prev;
    const amount = units * rate;

    records.push({ month, name, room, units, amount, paid: false });
    saveAndRender();

    document.querySelectorAll('.form-group input').forEach(i => i.value = '');
  }

  function togglePayment(i) {
    records[i].paid = !records[i].paid;
    saveAndRender();
  }

  function saveAndRender() {
    localStorage.setItem('records', JSON.stringify(records));
    renderTable();
    checkNotification();
  }

  function renderTable() {
    const tableBody = document.getElementById('tableBody');
    tableBody.innerHTML = '';
    records.forEach((r, i) => {
      tableBody.innerHTML += `
        <tr>
          <td>${i + 1}</td>
          <td>${r.month}</td>
          <td>${r.name}</td>
          <td>${r.room}</td>
          <td>${r.units}</td>
          <td>₹${r.amount.toFixed(2)}</td>
          <td class="${r.paid ? 'paid' : 'unpaid'}">${r.paid ? 'Paid' : 'Unpaid'}</td>
          <td><button onclick="togglePayment(${i})">Mark ${r.paid ? 'Unpaid' : 'Paid'}</button></td>
        </tr>`;
    });
  }

  function checkNotification() {
    const currentMonth = new Date().toISOString().slice(0,7);
    const lastNotified = localStorage.getItem('lastNotifiedMonth');
    const unpaid = records.filter(r => r.month === currentMonth && !r.paid);

    if (unpaid.length > 0 && lastNotified !== currentMonth) {
      alert(⚠️ Reminder: ${unpaid.length} unpaid electricity bills for ${currentMonth});
      localStorage.setItem('lastNotifiedMonth', currentMonth);
    }
  }

  function printBill() {
    window.print();
  }

  function shareWhatsApp() {
    const phone = '919532354278';
    const currentMonth = new Date().toISOString().slice(0,7);

    const monthRecords = records.filter(r => r.month === currentMonth);
    if (monthRecords.length === 0) {
      alert('No records for current month');
      return;
    }

    let message = 🧾 Electricity Bill – ${currentMonth}%0A%0A;
    monthRecords.forEach((r, i) => {
      message += ${i+1}. ${r.name} (Room ${r.room})%0AUnits: ${r.units}%0AAmount: ₹${r.amount}%0AStatus: ${r.paid ? 'Paid' : 'Unpaid'}%0A%0A;
    });
    message += '— Pankaj Mauray';

    window.open(https://wa.me/${phone}?text=${message}, '_blank');
  }

  renderTable();
  checkNotification();
</script>
